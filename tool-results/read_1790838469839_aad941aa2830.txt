    27→// Two shops can legally share a name — when they appear in the SAME feed,
    28→// suffix each with their area so buyers tap the right one. The shop's own
    29→// location wins (profile area → district), falling back to the listing's.
    30→function computeShopLabels(items: ListingsPage['items'] | undefined): Map<string, string> {
    31→  const labels = new Map<string, string>()
    32→  if (!items) return labels
    33→
    34→  const ownerIdsByName = new Map<string, Set<string>>()
    35→  for (const listing of items) {
    36→    const owner = listing.user
    37→    const name = owner?.profile?.businessName?.trim() || owner?.name
    38→    if (!owner || !name) continue
    39→    const key = normalizeShopName(name)
    40→    const ids = ownerIdsByName.get(key) ?? new Set<string>()
    41→    ids.add(owner.id)
    42→    ownerIdsByName.set(key, ids)
    43→  }
    44→
    45→  for (const listing of items) {
    46→    const owner = listing.user
    47→    const name = owner?.profile?.businessName?.trim() || owner?.name
    48→    if (!owner || !name) continue
    49→    const ids = ownerIdsByName.get(normalizeShopName(name))
    50→    if (!ids || ids.size < 2) continue
    51→    const area = owner.profile?.area || owner.profile?.county || listing.county
    52→    labels.set(listing.id, area ? `${name} · ${area}` : name)
    53→  }
    54→  return labels
    55→}

   356→// The Loved shelf — the shortlist, rendered as the same photo-first blocks
   357→// as the feed. Every item is re-fetched from the public API when the shelf
   358→// opens (the heart stores only an id), so what the buyer sees is what is
   359→// really there right now: sold-out items say so instead of pretending.
   360→function LovedShelf({
   361→  onOpen,
   362→  onOpenShop,
   363→  onAdd,
   364→  onBrowse,
   365→}: {
   366→  onOpen: (id: string) => void
   367→  onOpenShop: (shopId: string) => void
   368→  // Same shape ListingBlock hands every caller — blocks may come back from
   369→  // the public detail endpoint, but the add contract stays the wide one.
   370→  onAdd: (listing: Listing & { user?: ListingShopOwnerT }) => boolean | void
   371→  onBrowse: () => void
   372→}) {
   373→  const lovedIds = useLovedIds()
   374→
   375→  const itemsQuery = useQuery({
   376→    queryKey: ['loved-items', lovedIds.join(',')],
   377→    enabled: lovedIds.length > 0,
   378→    retry: false,
   379→    staleTime: 15_000,
   380→    queryFn: async () => {
   381→      const results = await Promise.all(
   382→        lovedIds.map(async (id) => {
   383→          try {
   384→            const res = await apiGet<{ listing: ListingDetailT }>(`/api/listings/${id}`)
   385→            return res.listing
   386→          } catch (err) {
   387→            // A 404 means the seller deleted it — a heart on a ghost is
   388→            // clutter that also eats a shortlist slot, so it unloves itself.
   389→            if (err instanceof Error && /404|does not exist|removed/i.test(err.message)) {
   390→              unlove(id)
   391→              return null
   392→            }
   393→            throw err
   394→          }
   395→        }),
   396→      )
   397→      return results.filter((l): l is ListingDetailT => l !== null)
   398→    },
   399→  })
   400→
   401→  if (lovedIds.length === 0) {
   402→    return (
   403→      <EmptyState
   404→        icon={<Heart />}
   405→        title="Nothing loved yet"
   406→        description="Tap the heart on any item and it waits here for later — like a name scribbled on a price list. It lives on this phone, just like the basket."
   407→        action={<Button onClick={onBrowse}>Browse listings</Button>}
   408→      />
   409→    )
   410→  }
   411→
   412→  if (itemsQuery.isLoading) {
   413→    return <ListingGridSkeleton />
   414→  }
   415→
   416→  if (itemsQuery.isError || !itemsQuery.data) {
   417→    return <ErrorState message="Could not load your loved items" onRetry={() => itemsQuery.refetch()} />
   418→  }
   419→
   420→  const items = itemsQuery.data
   421→  const available = items.filter((l) => l.status === 'ACTIVE')
   422→  const unavailable = items.filter((l) => l.status !== 'ACTIVE')
   423→
   424→  return (
   425→    <div className="space-y-3">
   426→      <p className="text-sm text-muted-foreground" role="status">
   427→        Your shortlist · {items.length} {items.length === 1 ? 'item' : 'items'}, newest first — lives on this phone
   428→      </p>
   429→
   430→      {available.length > 0 ? (
   431→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-3 sm:gap-3 lg:grid-cols-4">
   432→          {available.map((listing) => (
   433→            <ListingBlock
   434→              key={listing.id}
   435→              listing={listing}
   436→              onOpen={onOpen}
   437→              onOpenShop={onOpenShop}
   438→              onAdd={onAdd}
   439→            />
   440→          ))}
   441→        </div>
   442→      ) : null}
   443→
   444→      {/* Honest shelf: what expired or sold out is said out loud, with the
   445→          one action that fixes it. */}
   446→      {unavailable.length > 0 ? (
   447→        <div className="rounded-lg border border-amber-200 bg-amber-50/70 px-4 py-3">
   448→          <p className="flex items-start gap-1.5 text-xs font-medium text-amber-800">
   449→            <TriangleAlert className="mt-0.5 size-3.5 shrink-0" aria-hidden />
   450→            {unavailable.length === 1
   451→              ? '1 loved item is no longer available:'
   452→              : `${unavailable.length} loved items are no longer available:`}
   453→          </p>
   454→          <ul className="mt-1.5 space-y-1">
   455→            {unavailable.map((listing) => (
   456→              <li key={listing.id} className="flex items-center gap-2 text-sm">
   457→                <button
   458→                  type="button"
   459→                  onClick={() => onOpen(listing.id)}
   460→                  className="min-w-0 flex-1 truncate text-left hover:underline"
   461→                >
   462→                  {listing.title}
   463→                </button>
   464→                <HeartButton listingId={listing.id} title={listing.title} className="size-7 shrink-0 border-0 bg-transparent hover:bg-transparent" />
   465→                <span className="shrink-0 text-xs text-amber-800">tap the heart to forget it</span>
   466→              </li>
   467→            ))}
   468→          </ul>
   469→        </div>
   470→      ) : null}
   471→    </div>
   472→  )
   473→}

   475→// The result of punching a DK code into search — one shop, whole card taps
   476→// through, same affordance as a listing card. The code chip repeats so the
   477→// buyer can confirm the number they typed matches the shop they got.
   478→function ShopCodeCard({ shop, onOpen }: { shop: ShopLookupResponse['shop']; onOpen: () => void }) {
   479→  return (
   480→    <button
   481→      type="button"
   482→      onClick={onOpen}

   484→      aria-label={`Open shop: ${shop.name}`}
   485→    >
   486→      {shop.photoUrl ? (
   487→        <img src={shop.photoUrl} alt="" className="size-14 shrink-0 rounded-lg border object-cover" />
   488→      ) : (
   489→        <span className="flex size-14 shrink-0 items-center justify-center rounded-lg border bg-accent text-xl font-bold text-accent-foreground">
   490→          {shop.name.charAt(0).toUpperCase()}
   491→        </span>

   650→function SaveSearchButton() {
   651→  const { filters } = useAppStore()
   652→  const { user } = useSession()
   653→  const setAuthOpen = useAppStore((s) => s.setAuthOpen)
   654→  const { toast } = useToast()
   655→  const queryClient = useQueryClient()
   656→  const [open, setOpen] = useState(false)
   657→  const [name, setName] = useState('')
   658→  const [busy, setBusy] = useState(false)
   659→
   660→  function openDialog() {
   661→    if (!user) {
   662→      setAuthOpen(true)
   663→      return
   664→    }
   665→    // Suggest an honest, readable name from current filters.
   666→    const bits: string[] = []
   667→    if (filters.type !== 'any') bits.push(filters.type === 'OFFER' ? 'Offers' : 'Requests')
   668→    if (filters.category !== 'any') bits.push(CATEGORIES.find((c) => c.key === filters.category)?.label ?? filters.category)
   669→    if (filters.county !== 'any') bits.push(filters.county)
   670→    if (filters.q) bits.push(`"${filters.q}"`)
   671→    setName(bits.join(' · ') || 'Everything')
   672→    setOpen(true)
   673→  }
   674→
   675→  async function save() {
   676→    if (!name.trim()) return
   677→    setBusy(true)
   678→    try {
   679→      await apiPost('/api/saved-searches', {
   680→        name: name.trim(),
   681→        query: {
   682→          q: filters.q || undefined,
   683→          type: filters.type !== 'any' ? filters.type : undefined,
   684→          category: filters.category !== 'any' ? filters.category : undefined,
   685→          county: filters.county !== 'any' ? filters.county : undefined,
   686→          unit: filters.unit !== 'any' ? filters.unit : undefined,
   687→          minPrice: filters.minPrice !== '' ? Number(filters.minPrice) : undefined,
   688→          maxPrice: filters.maxPrice !== '' ? Number(filters.maxPrice) : undefined,
   689→        },
   690→      })
   691→      await queryClient.invalidateQueries({ queryKey: ['saved-searches'] })
   692→      toast({ title: 'Search saved', description: 'You will get alerts when new listings match.' })
   693→      setOpen(false)
   694→    } catch (err) {
   695→      toast({ title: 'Could not save search', description: err instanceof Error ? err.message : undefined, variant: 'destructive' })
   696→    } finally {
   697→      setBusy(false)
   698→    }
   699→  }
   700→
   701→  return (
   702→    <>
   703→      <Button type="button" variant="outline" size="sm" className="shrink-0 gap-1.5" onClick={openDialog}>
   704→        <Bookmark className="size-3.5" aria-hidden /> Save this search
   705→      </Button>
   706→      <Dialog open={open} onOpenChange={setOpen}>
   707→        <DialogContent className="sm:max-w-sm">
   708→          <DialogHeader>
   709→            <DialogTitle>Save this search</DialogTitle>
   710→            <DialogDescription>We will alert you when new listings match these filters.</DialogDescription>
   711→          </DialogHeader>
   712→          <div className="space-y-1.5">
   713→            <Label htmlFor="ss-name">Search name</Label>
   714→            <Input id="ss-name" value={name} onChange={(e) => setName(e.target.value)} maxLength={60} />
   715→          </div>
   716→          <DialogFooter>
   717→            <Button type="button" onClick={save} disabled={busy || !name.trim()}>
   718→              {busy ? 'Saving…' : 'Save search'}
   719→            </Button>
   720→          </DialogFooter>
   721→        </DialogContent>
   722→      </Dialog>
   723→    </>
   724→  )
   725→}
